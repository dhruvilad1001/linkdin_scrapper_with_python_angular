class ProfileManager:
    """Handles all profile-related database operations"""
    
    def __init__(self, collection):
        self.collection = collection
    
    def get_all_profiles(self, limit=50, skip=0):
        """Get all profiles with pagination"""
        try:
            if self.collection is None:
                return []
                
            profiles = list(self.collection.find({}, {
                'name': 1, 'headline': 1, 'url': 1, 'last_updated': 1,
                'experience': {'$slice': 1}, 'education': {'$slice': 1}
            }).sort('last_updated', -1).skip(skip).limit(limit))
            
            for profile in profiles:
                profile['_id'] = str(profile['_id'])
                if 'last_updated' in profile and profile['last_updated']:
                    if isinstance(profile['last_updated'], datetime):
                        profile['last_updated'] = profile['last_updated'].strftime('%Y-%m-%d %H:%M:%S')
            
            return profiles
        except Exception as e:
            logger.error(f"Error fetching profiles: {e}")
            return []
    
    def get_profile_by_id(self, profile_id):
        """Get detailed profile by ID"""
        try:
            if self.collection is None:
                return None
                
            profile = self.collection.find_one({'_id': ObjectId(profile_id)})
            if profile:
                profile['_id'] = str(profile['_id'])
                if 'last_updated' in profile and profile['last_updated']:
                    if isinstance(profile['last_updated'], datetime):
                        profile['last_updated'] = profile['last_updated'].strftime('%Y-%m-%d %H:%M:%S')
            return profile
        except Exception as e:
            logger.error(f"Error fetching profile {profile_id}: {e}")
            return None
    
    def search_profiles(self, query, limit=20):
        """Search profiles by name, headline, company, or college..."""
        try:
            if self.collection is None:
                return []
                
            search_filter = {
                '$or': [
                    {'name': {'$regex': query, '$options': 'i'}},
                    {'headline': {'$regex': query, '$options': 'i'}},
                    {'experience.company_name': {'$regex': query, '$options': 'i'}},    
                    {'education.college': {'$regex': query, '$options': 'i'}},
                    # {'education.degree': {'$regex': query, '$options': 'i'}}
                ]
            }
            
            profiles = list(self.collection.find(search_filter, {
                'name': 1, 'headline': 1, 'url': 1, 'last_updated': 1, 'experience': 1, 'education': 1
            }).limit(limit))
            
            for profile in profiles:
                profile['_id'] = str(profile['_id'])
            
            return profiles
        except Exception as e:
            logger.error(f"Error searching profiles: {e}")
            return []
    
    def get_statistics(self):
        """Get database statistics"""
        try:
            if self.collection is None:
                return {'total_profiles': 0, 'today_profiles': 0, 'top_companies': [], 'top_schools': []}
                
            total_profiles = self.collection.count_documents({})
            
            today_start = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
            today_profiles = self.collection.count_documents({
                'last_updated': {'$gte': today_start}
            })
            
            company_pipeline = [
                {'$unwind': '$experience'},
                {'$group': {'_id': '$experience.company_name', 'count': {'$sum': 1}}},
                {'$sort': {'count': -1}},
                {'$limit': 10}
            ]
            top_companies = list(self.collection.aggregate(company_pipeline))
            
            education_pipeline = [
                {'$unwind': '$education'},
                {'$group': {'_id': '$education.college', 'count': {'$sum': 1}}},
                {'$sort': {'count': -1}},
                {'$limit': 10}
            ]
            top_schools = list(self.collection.aggregate(education_pipeline))
            
            return {
                'total_profiles': total_profiles,
                'today_profiles': today_profiles,
                'top_companies': top_companies,
                'top_schools': top_schools
            }
        except Exception as e:
            logger.error(f"Error getting statistics: {e}")
            return {'total_profiles': 0, 'today_profiles': 0, 'top_companies': [], 'top_schools': []}
    
    def delete_profile(self, profile_id):
        """Delete a profile by ID"""
        try:
            if self.collection is None:
                return False
                
            result = self.collection.delete_one({'_id': ObjectId(profile_id)})
            return result.deleted_count > 0
        except Exception as e:
            logger.error(f"Error deleting profile {profile_id}: {e}")
            return False

class ScrapingManager:
    """Handles scraping operations"""
    
    @staticmethod
    def validate_linkedin_url(url):
        """Validate if URL is a valid LinkedIn profile URL"""
        pattern = r'^https:\/\/www\.linkedin\.com\/in\/[a-zA-Z0-9\-]+\/?$'
        return re.match(pattern, url) is not None
    
    @staticmethod
    def run_scraper(url):
        """Run the actual scraper for a given URL"""
        try:
            result = run_linkedin_scraper(url)
            if isinstance(result, tuple) and len(result) >= 2:
                success, message = result[0], result[1]
            elif isinstance(result, bool):
                success, message = result, 'OK' if result else 'Failed'
            else:
                success, message = False, f'Unexpected scraper return: {repr(result)}'
            return success, message
        except Exception as e:
            logger.error(f"run_scraper exception: {e}")
            return False, str(e)

class ExportManager:
    """Handles data export operations with proper formatting"""
    
    def __init__(self, collection):
        self.collection = collection
    
    def clean_text(self, text):
        """Clean text data"""
        if text is None:
            return ""
        s = str(text)
        replacements = {
            'Â·': '·', '·': '-', '•': '•', '\xa0': ' ', '\u200b': '',
            '\r\n': ' ', '\r': ' ', '\n': ' '
        }
        for k, v in replacements.items():
            s = s.replace(k, v)
        s = re.sub(r'[\x00-\x1F\x7F]+', ' ', s)
        s = re.sub(r'\s+', ' ', s).strip()
        return s
    
    def extract_dates(self, duration_str):
        """Extract dates from duration string"""
        if not duration_str:
            return {"duration": "", "start_date": "", "end_date": "", "duration_months": ""}
        
        s = self.clean_text(duration_str)
        pattern = r'(\w{3}\s+\d{4})\s+to\s+((?:\w{3}\s+\d{4})|Present)\s*·?\s*(.+)?'
        match = re.search(pattern, s, re.IGNORECASE)
        
        if match:
            duration_text = self.clean_text(match.group(3) or "")
            duration_months = self.convert_duration_to_months(duration_text)
            
            return {
                "start_date": match.group(1),
                "end_date": match.group(2),
                "duration": duration_text,
                "duration_months": duration_months
            }
        
        # If no match, try to extract duration directly
        duration_months = self.convert_duration_to_months(s)
        return {"duration": s, "start_date": "", "end_date": "", "duration_months": duration_months}
    
    def convert_duration_to_months(self, duration_str):
        """Convert duration string to total months"""
        if not duration_str:
            return ""
        
        s = duration_str.lower()
        total_months = 0
        
        # Extract years
        year_match = re.search(r'(\d+)\s*yr', s)
        if year_match:
            total_months += int(year_match.group(1)) * 12
        
        # Extract months
        month_match = re.search(r'(\d+)\s*mo', s)
        if month_match:
            total_months += int(month_match.group(1))
        
        return str(total_months) if total_months > 0 else ""
    
    def extract_location(self, location_str):
        """Extract location details"""
        if not location_str:
            return {"city": "", "state": "", "country": "", "work_mode": ""}
        
        s = self.clean_text(location_str)
        work_mode = ""
        
        for mode in ["Remote", "On-site", "Hybrid", "Onsite"]:
            if mode.lower() in s.lower():
                work_mode = mode
                s = s.replace(mode, "").strip()
        
        s = re.sub(r'\s*·\s*', '', s).strip()
        parts = [p.strip() for p in s.split(',') if p.strip()]
        
        return {
            "city": parts[0] if len(parts) > 0 else "",
            "state": parts[1] if len(parts) > 1 else "",
            "country": parts[2] if len(parts) > 2 else (parts[-1] if parts else ""),
            "work_mode": work_mode
        }
    
    def export_to_excel(self, format_type='grouped', filters=None):
        """Export profiles to Excel with proper formatting"""
        try:
            if self.collection is None:
                return None, "Database not connected"
            
            query = filters if filters else {}
            profiles = list(self.collection.find(query))
            
            if not profiles:
                return None, "No profiles found to export"
            
            if format_type == 'grouped':
                return self._export_grouped_format(profiles)
            else:
                return self._export_flat_format(profiles)
                
        except Exception as e:
            logger.error(f"Error exporting to Excel: {e}")
            return None, str(e)

    def _export_flat_format(self, profiles):
        """Export in flat format (single sheet)"""
        try:
            filename = f"linkedin_profiles_flat_{datetime.now().strftime('%Y%m%d_%H%M%S')}.xlsx"
            filepath = os.path.join('exports', filename)
            
            flat_data = []
            for profile in profiles:
                first_exp = profile.get('experience', [{}])[0] if profile.get('experience') else {}
                first_edu = profile.get('education', [{}])[0] if profile.get('education') else {}
                
                # Calculate total experience in months
                total_months = 0
                for exp in profile.get('experience', []):
                    dates = self.extract_dates(exp.get('duration', ''))
                    if dates['duration_months']:
                        total_months += int(dates['duration_months'])
                
                flat_data.append({
                    'Name': profile.get('name', ''),
                    'URL': profile.get('url', ''),
                    'Headline': profile.get('headline', ''),
                    'About': profile.get('about', '')[:300] if profile.get('about') else '',
                    'Current_Company': first_exp.get('company_name', ''),
                    'Current_Position': first_exp.get('designation', ''),
                    'Total_Experience_Months': total_months,
                    'Education_College': first_edu.get('college', ''),
                    'Education_Degree': first_edu.get('degree', ''),
                    'Total_Projects': len(profile.get('projects', [])),
                    'Total_Certifications': len(profile.get('licenses', [])),
                    'Last_Updated': profile.get('last_updated', '')
                })
            
            df = pd.DataFrame(flat_data)
            df.to_excel(filepath, index=False, engine='openpyxl')
            
            return filepath, None
            
        except Exception as e:
            return None, str(e)

# Helper function for analytics
def calculate_profile_completeness(profile):
    """Calculate profile completeness percentage"""
    score = 0
    total_points = 10
    
    # Name (mandatory)
    if profile.get('name') and profile.get('name') != 'N/A':
        score += 1
    
    # Headline
    if profile.get('headline') and profile.get('headline') != 'N/A':
        score += 1
    
    # About section
    if profile.get('about') and profile.get('about') != 'N/A' and len(profile.get('about', '')) > 50:
        score += 1
    
    # Experience (at least one)
    if profile.get('experience') and len(profile.get('experience', [])) > 0:
        score += 2
    
    # Education (at least one)
    if profile.get('education') and len(profile.get('education', [])) > 0:
        score += 2
    
    # Projects
    if profile.get('projects') and len(profile.get('projects', [])) > 0:
        score += 1
    
    # Certifications
    if profile.get('licenses') and len(profile.get('licenses', [])) > 0:
        score += 1
    
    # Courses or Awards
    if (profile.get('courses') and len(profile.get('courses', [])) > 0) or \
       (profile.get('honors_awards') and len(profile.get('honors_awards', [])) > 0):
        score += 1
    
    return round((score / total_points) * 100, 1)

    try:

        url = data.get('url', '').strip()

            return jsonify({'success': False, 'error': 'URL is required'}), 400

                    'url': url,

                new_profile_id = str(new_profile['_id']) if new_profile else None
                
                return jsonify({
                    'success': True, 
                    'message': 'Profile scraped and saved successfully',
                    'url': url,
                    'action': 'created',
                    'profile_id': new_profile_id
                })
        else:
            return jsonify({
                'success': False,
                'error': message
            }), 500
        
    except Exception as e:
        logger.error(f"API scrape error: {e}")
        return jsonify({'success': False, 'error': str(e)}), 500

@app.route('/api/bulk-scrape', methods=['POST'])
def api_bulk_scrape():
    """API endpoint for bulk scraping"""
    try:
        data = request.get_json()
        urls = data.get('urls', [])
        
        if not urls:
            return jsonify({'success': False, 'error': 'No URLs provided'}), 400
        
        valid_urls = []
        invalid_urls = []
        existing_urls = []
        
        for url in urls:
            url = url.strip()
            if ScrapingManager.validate_linkedin_url(url):
                # Don't skip existing URLs anymore - they will be updated
                valid_urls.append(url)
                if collection is not None and collection.find_one({'url': url}):
                    existing_urls.append(url)
            else:
                invalid_urls.append(url)
        
        if not valid_urls:
            return jsonify({
                'success': False, 
                'error': 'No valid URLs to scrape',
                'invalid_urls': invalid_urls
            }), 400
        
        def bulk_scrape_background():
            for url in valid_urls:
                try:
                    success, message = ScrapingManager.run_scraper(url)
                    if success:
                        logger.info(f"Bulk scraped: {url}")
                    else:
                        logger.error(f"Failed bulk scraping {url}: {message}")
                except Exception as e:
                    logger.error(f"Bulk scraping error for {url}: {e}")
        
        thread = threading.Thread(target=bulk_scrape_background)
        thread.daemon = True
        thread.start()
        
        return jsonify({
            'success': True,
            'message': f'Bulk scraping started for {len(valid_urls)} profiles ({len(existing_urls)} will be updated)',
            'valid_urls': valid_urls,
            'existing_urls': existing_urls,
            'invalid_urls': invalid_urls
        })
        
    except Exception as e:
        logger.error(f"API bulk scrape error: {e}")
        return jsonify({'success': False, 'error': str(e)}), 500

@app.template_filter('datetime_format')
def datetime_format(value, format='%Y-%m-%d %H:%M'):
    if isinstance(value, str):
        try:
            value = datetime.strptime(value, '%Y-%m-%d %H:%M:%S')
        except:
            return value
    return value.strftime(format) if value else ''

@app.template_filter('truncate_text')
def truncate_text(text, length=100):
    if not text:
        return ''
    return text[:length] + '...' if len(text) > length else text

@app.template_filter('format_number')
def format_number(value):
    try:
        return f"{int(value):,}"
    except:
        return value

def run_linkedin_scraper(profile_url):
    """
    Scrapes a LinkedIn profile (Name, Headline, About, Experience, Education, Projects)
    Saves to MongoDB and exports to Excel.
    """
    try:
        # --- MongoDB Connection ---
        client = MongoClient(os.getenv("MONGO_URI", "mongodb://localhost:27017/"))
        db = client[os.getenv("DATABASE_NAME", "Linkedin")]
        collection = db[os.getenv("COLLECTION_NAME", "profile_data")]

        # --- Chrome Options ---
        chrome_options = Options()
        chrome_options.add_argument("--headless=new")
        chrome_options.add_argument("--no-sandbox")
        chrome_options.add_argument("--disable-dev-shm-usage")
        chrome_options.add_argument("--disable-blink-features=AutomationControlled")
        chrome_options.add_argument("--window-size=1920,1080")

        driver = webdriver.Chrome(options=chrome_options)
        wait = WebDriverWait(driver, 15)

        try:
            # --- Login Phase ---
            driver.get("https://www.linkedin.com/login")
            sleep(2)

            email = os.getenv("Email")
            password = os.getenv("Password")

            if not email or not password:
                raise ValueError("LinkedIn Email or Password not found in .env file")

            driver.find_element(By.ID, "username").send_keys(email)
            driver.find_element(By.ID, "password").send_keys(password, Keys.RETURN)
            sleep(5)

            # --- Scrape Target Profile ---
            driver.get(profile_url)
            wait.until(EC.presence_of_element_located((By.TAG_NAME, "body")))
            sleep(3)

            driver.execute_script("window.scrollTo(0, document.body.scrollHeight);")
            sleep(2)

            # Parse HTML with BeautifulSoup for easier extraction
            soup = BeautifulSoup(driver.page_source, "html.parser")

            # --- Name ---
            try:
                name = soup.find("h1").get_text(strip=True)
            except:
                name = "N/A"

            # --- Headline ---
            try:
                headline = soup.find("div", class_="text-body-medium").get_text(strip=True)
            except:
                headline = "N/A"

            # --- About Section ---
            try:
                about_tag = soup.find("div", {"id": "about"}).find_next("span")
                about = about_tag.get_text(strip=True)
            except:
                about = "N/A"

            # --- Experience Section ---
            experience = []
            try:
                exp_section = soup.find("section", {"id": "experience"})
                if exp_section:
                    exp_items = exp_section.find_all("li")
                    for exp in exp_items[:5]:
                        title = exp.find("span", {"class": "mr1 t-bold"}).get_text(strip=True) if exp.find("span", {"class": "mr1 t-bold"}) else ""
                        company = exp.find("span", {"class": "t-14 t-normal"}).get_text(strip=True) if exp.find("span", {"class": "t-14 t-normal"}) else ""
                        duration = exp.find("span", {"class": "t-14 t-normal t-black--light"}).get_text(strip=True) if exp.find("span", {"class": "t-14 t-normal t-black--light"}) else ""
                        experience.append({
                            "title": title,
                            "company": company,
                            "duration": duration
                        })
            except Exception as e:
                logger.warning(f"⚠️ Experience not extracted: {e}")

            # --- Education Section ---
            education = []
            try:
                edu_section = soup.find("section", {"id": "education"})
                if edu_section:
                    edu_items = edu_section.find_all("li")
                    for edu in edu_items[:3]:
                        school = edu.find("span", {"class": "mr1 hoverable-link-text"}).get_text(strip=True) if edu.find("span", {"class": "mr1 hoverable-link-text"}) else ""
                        degree = edu.find("span", {"class": "t-14 t-normal"}).get_text(strip=True) if edu.find("span", {"class": "t-14 t-normal"}) else ""
                        education.append({
                            "school": school,
                            "degree": degree
                        })
            except Exception as e:
                logger.warning(f"⚠️ Education not extracted: {e}")

            # --- Projects Section ---
            projects = []
            try:
                proj_section = soup.find("section", {"id": "projects"})
                if proj_section:
                    proj_items = proj_section.find_all("li")
                    for proj in proj_items[:3]:
                        project_title = proj.find("span", {"class": "mr1 t-bold"}).get_text(strip=True) if proj.find("span", {"class": "mr1 t-bold"}) else ""
                        project_desc = proj.find("span", {"class": "t-14 t-normal"}).get_text(strip=True) if proj.find("span", {"class": "t-14 t-normal"}) else ""
                        projects.append({
                            "project_title": project_title,
                            "project_desc": project_desc
                        })
            except Exception as e:
                logger.warning(f"⚠️ Projects not extracted: {e}")

            # --- Prepare and Save Data ---
            profile_data = {
                "name": name,
                "url": profile_url,
                "headline": headline,
                "about": about,
                "experience": experience,
                "education": education,
                "projects": projects,
                "last_updated": datetime.now(),
                "scraped_at": datetime.now()

                "Name": name,
                "URL": profile_url,
                "Headline": headline,
                "About": about,
                "Experience": str(experience),
                "Education": str(education),
                "Projects": str(projects),
                "Last Updated": datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            }])

            if not os.path.exists(excel_file):
                df.to_excel(excel_file, index=False)
            else:
                existing_df = pd.read_excel(excel_file)
                combined = pd.concat([existing_df, df], ignore_index=True)
                combined.to_excel(excel_file, index=False)

            logger.info(f"✅ Profile scraped and exported: {profile_url}")
            return True, "Profile scraped successfully"

        finally:
            driver.quit()

    except Exception as e:
        logger.error(f"❌ Error scraping {profile_url}: {e}")
        return False, str(e)

    finally:
        if "client" in locals():
            client.close()

def get_experience_tokens(driver, timeout=15):
    """Scroll and fetch all hidden text tokens from the Experience section."""
    driver.execute_script("window.scrollTo(0, document.body.scrollHeight * 0.35);")
    sleep(2)

    try:
        exp_section = WebDriverWait(driver, timeout).until(
            EC.presence_of_element_located((By.XPATH, "//section[.//span[text()='Experience']]"))
        )
    except TimeoutException:
        logger.info("ℹ️ No 'Experience' section found on this profile.")
        return []

    exp_html = exp_section.get_attribute("outerHTML")
    exp_soup = BeautifulSoup(exp_html, "lxml")
    tokens = [s.get_text(strip=True) for s in exp_soup.select("span.visually-hidden") if s.get_text(strip=True)]
    return tokens


def parse_experience_tokens(tokens):
    """Parse tokens into clean company_name, designation, duration, location"""
    if tokens and tokens[0].strip().lower() == "experience":
        tokens = tokens[1:]

    roles_set = {
        "Full-time", "Part-time", "Internship", "Contract", "Self-employed",
        "Freelance", "Apprenticeship", "Volunteer", "Temporary", "Seasonal"
    }

    def is_duration(s: str) -> bool:
        s2 = s.lower()
        return (" to " in s2 and ("present" in s2 or "mo" in s2 or "yr" in s2)) or (" yr" in s2) or (" mo" in s2)

    def is_location(s: str) -> bool:
        if is_duration(s): return False
        if s in roles_set: return False
        loc_keywords = ["india", "remote", "on-site", "hybrid", "onsite", "work from home"]
        if any(k in s.lower() for k in loc_keywords): return True
        return ("," in s and len(s.split()) >= 2)

    def clean_company(s: str) -> str:
        if not s: return ""
        for r in roles_set:
            s = s.replace("· " + r, "").replace(r, "")
        return s.strip()

    def looks_like_description(s: str, is_company=False) -> bool:
        if is_company: return False
        return (len(s.split()) > 15 or len(s) > 120 or s.lower().endswith((".png", ".jpg", ".jpeg")))

    result = []
    i, n = 0, len(tokens)

    while i < n:
        designation, company, dur, loc = "", "", "", ""

        if i < n:
            designation = tokens[i]; i += 1

        if i < n:
            if tokens[i] not in roles_set and not is_duration(tokens[i]) and not is_location(tokens[i]):
                company = tokens[i]; i += 1

        if i < n and is_duration(tokens[i]):
            dur = tokens[i]; i += 1

        if i < n and is_location(tokens[i]):
            loc = tokens[i]; i += 1

        # Fix: swap if designation looks like duration
        if is_duration(designation):
            dur, designation = designation, ""

        if looks_like_description(designation):
            designation = ""

        if company and any(kw in company.lower() for kw in ["pvt", "inc", "solutions", "system", "hr", "ltd"]):
            pass
        elif looks_like_description(company, is_company=True):
            company = ""

        if not designation and not company:
            continue

        result.append({
            "company_name": clean_company(company),
            "designation": designation,
            "duration": dur,
            "location": loc
        })

    clean_result = [r for r in result if not looks_like_description(r["duration"])]
    return clean_result

def get_experience_tokens(driver, timeout=15):
    """Scroll and fetch all hidden text tokens from the Experience section."""
    driver.execute_script("window.scrollTo(0, document.body.scrollHeight * 0.35);")
    sleep(2)

    try:
        exp_section = WebDriverWait(driver, timeout).until(
            EC.presence_of_element_located((By.XPATH, "//section[.//span[text()='Experience']]"))
        )
    except TimeoutException:
        logger.info("ℹ️ No 'Experience' section found on this profile.")
        return []

    exp_html = exp_section.get_attribute("outerHTML")
    exp_soup = BeautifulSoup(exp_html, "lxml")
    tokens = [s.get_text(strip=True) for s in exp_soup.select("span.visually-hidden") if s.get_text(strip=True)]
    return tokens


def parse_experience_tokens(tokens):
    """Parse tokens into clean company_name, designation, duration, location"""
    if tokens and tokens[0].strip().lower() == "experience":
        tokens = tokens[1:]

    roles_set = {
        "Full-time", "Part-time", "Internship", "Contract", "Self-employed",
        "Freelance", "Apprenticeship", "Volunteer", "Temporary", "Seasonal"
    }

    def is_duration(s: str) -> bool:
        s2 = s.lower()
        return (" to " in s2 and ("present" in s2 or "mo" in s2 or "yr" in s2)) or (" yr" in s2) or (" mo" in s2)

    def is_location(s: str) -> bool:
        if is_duration(s): return False
        if s in roles_set: return False
        loc_keywords = ["india", "remote", "on-site", "hybrid", "onsite", "work from home"]
        if any(k in s.lower() for k in loc_keywords): return True
        return ("," in s and len(s.split()) >= 2)

    def clean_company(s: str) -> str:
        if not s: return ""
        for r in roles_set:
            s = s.replace("· " + r, "").replace(r, "")
        return s.strip()

    def looks_like_description(s: str, is_company=False) -> bool:
        if is_company: return False
        return (len(s.split()) > 15 or len(s) > 120 or s.lower().endswith((".png", ".jpg", ".jpeg")))

    result = []
    i, n = 0, len(tokens)

    while i < n:
        designation, company, dur, loc = "", "", "", ""

        if i < n:
            designation = tokens[i]; i += 1

        if i < n:
            if tokens[i] not in roles_set and not is_duration(tokens[i]) and not is_location(tokens[i]):
                company = tokens[i]; i += 1

        if i < n and is_duration(tokens[i]):
            dur = tokens[i]; i += 1

        if i < n and is_location(tokens[i]):
            loc = tokens[i]; i += 1

        # Fix: swap if designation looks like duration
        if is_duration(designation):
            dur, designation = designation, ""

        if looks_like_description(designation):
            designation = ""

        if company and any(kw in company.lower() for kw in ["pvt", "inc", "solutions", "system", "hr", "ltd"]):
            pass
        elif looks_like_description(company, is_company=True):
            company = ""

        if not designation and not company:
            continue

        result.append({
            "company_name": clean_company(company),
            "designation": designation,
            "duration": dur,
            "location": loc
        })

    clean_result = [r for r in result if not looks_like_description(r["duration"])]
    return clean_result

                        items = lic_soup.find_all("li", {"class": "pvs-list__paged-list-item"})
                else:
                    items = []

                licenses = []
                for item in items:
                    spans = item.find_all('span', {'class': 'visually-hidden'})
                    filtered_spans = [
                        s.get_text(strip=True) for s in spans
                        if not any(keyword in s.get_text(strip=True).lower() 
                                 for keyword in ['connection', 'mutual', 'profile', 'view'])
                    ]

                    lic_dict = {}
                    if len(filtered_spans) > 0:
                        lic_dict['name'] = filtered_spans[0]
                    if len(filtered_spans) > 1:
                        lic_dict['institute'] = filtered_spans[1]
                    if len(filtered_spans) > 2:
                        lic_dict['issued_date'] = filtered_spans[2]

                    if lic_dict.get('name'):
                        licenses.append(lic_dict)
            except Exception as e:
                logger.warning(f"⚠️ Licenses error: {e}")
                licenses = []

            # --- COURSES ---
            try:
                courses_section = driver.find_elements(By.XPATH, 
                    "//section[.//span[contains(text(), 'Course')]]")
                
                courses = []
                if courses_section:
                    courses_html = courses_section[0].get_attribute("outerHTML")
                    courses_soup = BeautifulSoup(courses_html, "lxml")
                    items = courses_soup.find_all("li", {"class": "artdeco-list__item"})
                    if not items:
                        items = courses_soup.find_all("li", {"class": "pvs-list__paged-list-item"})

                    for item in items:
                        spans = item.find_all('span', {'class': 'visually-hidden'})
                        filtered_spans = [
                            s.get_text(strip=True) for s in spans
                            if not any(keyword in s.get_text(strip=True).lower() 
                                     for keyword in ['connection', 'mutual', 'profile'])
                        ]

                        if len(filtered_spans) >= 2:
                            courses.append({
                                'course_name': filtered_spans[0],
                                'associated_with': filtered_spans[1]
                            })
            except Exception as e:
                logger.warning(f"⚠️ Courses error: {e}")
                courses = []

            # --- HONORS & AWARDS ---
            try:
                awards_section = driver.find_elements(By.XPATH, 
                    "//section[.//span[contains(text(), 'Honors') or contains(text(), 'awards')]]")
                
                honors_awards = []
                if awards_section:
                    awards_html = awards_section[0].get_attribute("outerHTML")
                    awards_soup = BeautifulSoup(awards_html, "lxml")
                    items = awards_soup.find_all("li", {"class": "artdeco-list__item"})
                    if not items:
                        items = awards_soup.find_all("li", {"class": "pvs-list__paged-list-item"})

                    for item in items:
                        spans = [s.get_text(strip=True) for s in item.find_all("span", {"class": "visually-hidden"})]
                        filtered_spans = [
                            span for span in spans
                            if not any(x in span.lower() for x in ["connection", "studied at", "view profile"])
                        ]

                        if filtered_spans:
                            award_dict = {}
                            if len(filtered_spans) >= 1:
                                award_dict['award_name'] = filtered_spans[0]
                            if len(filtered_spans) >= 2:
                                parts = filtered_spans[1].split('·')
                                award_dict['issued_by'] = parts[0].replace('Issued by', '').strip()
                                if len(parts) > 1:
                                    award_dict['issued_date'] = parts[1].strip()
                            if len(filtered_spans) >= 3:
                                award_dict['associated_with'] = filtered_spans[2].replace('Associated with', '').strip()

                            honors_awards.append(award_dict)
            except Exception as e:
                logger.warning(f"⚠️ Awards error: {e}")
                honors_awards = []

            # --- PREPARE PROFILE DATA ---
            profile_data = {
                "name": name,
                "url": profile_url,
                "headline": headline,
                "about": about,
                "experience": experience,
                "education": education,
                "projects": projects,
                "licenses": licenses,
                "courses": courses,
                "honors_awards": honors_awards,
                "last_updated": datetime.now()
            }
            
            # Add scraped_at only for new profiles
            if not is_update:
                profile_data["scraped_at"] = datetime.now()

            # --- SAVE OR UPDATE TO MONGODB ---
            if is_update:
                # Update existing profile
                collection.update_one(
                    {"url": profile_url},
                    {"$set": profile_data}
                )
                logger.info(f"🔄 Profile updated: {name}")
                return True, "Profile updated successfully"
            else:
                # Insert new profile
                collection.insert_one(profile_data)
                logger.info(f"✅ Profile saved: {name}")
                return True, "Profile scraped successfully"

        finally:
            driver.quit()

    except Exception as e:
        logger.error(f"❌ Error scraping {profile_url}: {e}")
        return False, str(e)

    finally:
        if "client" in locals():
            client.close()
