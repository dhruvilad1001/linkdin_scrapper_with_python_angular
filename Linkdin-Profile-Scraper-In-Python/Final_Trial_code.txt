class ProfileManager:
    """Handles all profile-related database operations"""
    
    def __init__(self, collection):
        self.collection = collection
    
    def get_all_profiles(self, limit=50, skip=0):
        """Get all profiles with pagination"""
        try:
            if self.collection is None:
                return []
                
            profiles = list(self.collection.find({}, {
                'name': 1, 'headline': 1, 'url': 1, 'last_updated': 1,
                'experience': {'$slice': 1}, 'education': {'$slice': 1}
            }).sort('last_updated', -1).skip(skip).limit(limit))
            
            for profile in profiles:
                profile['_id'] = str(profile['_id'])
                if 'last_updated' in profile and profile['last_updated']:
                    if isinstance(profile['last_updated'], datetime):
                        profile['last_updated'] = profile['last_updated'].strftime('%Y-%m-%d %H:%M:%S')
            
            return profiles
        except Exception as e:
            logger.error(f"Error fetching profiles: {e}")
            return []
    
    def get_profile_by_id(self, profile_id):
        """Get detailed profile by ID"""
        try:
            if self.collection is None:
                return None
                
            profile = self.collection.find_one({'_id': ObjectId(profile_id)})
            if profile:
                profile['_id'] = str(profile['_id'])
                if 'last_updated' in profile and profile['last_updated']:
                    if isinstance(profile['last_updated'], datetime):
                        profile['last_updated'] = profile['last_updated'].strftime('%Y-%m-%d %H:%M:%S')
            return profile
        except Exception as e:
            logger.error(f"Error fetching profile {profile_id}: {e}")
            return None
    
    def search_profiles(self, query, limit=20):
        """Search profiles by name, headline, or company"""
        try:
            if self.collection is None:
                return []
                
            search_filter = {
                '$or': [
                    {'name': {'$regex': query, '$options': 'i'}},
                    {'headline': {'$regex': query, '$options': 'i'}},
                    {'experience.company_name': {'$regex': query, '$options': 'i'}}
                ]
            }
            
            profiles = list(self.collection.find(search_filter, {
                'name': 1, 'headline': 1, 'url': 1, 'last_updated': 1
            }).limit(limit))
            
            for profile in profiles:
                profile['_id'] = str(profile['_id'])
            
            return profiles
        except Exception as e:
            logger.error(f"Error searching profiles: {e}")
            return []
    
    def get_statistics(self):
        """Get database statistics"""
        try:
            if self.collection is None:
                return {'total_profiles': 0, 'today_profiles': 0, 'top_companies': [], 'top_schools': []}
                
            total_profiles = self.collection.count_documents({})
            
            today_start = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
            today_profiles = self.collection.count_documents({
                'last_updated': {'$gte': today_start}
            })
            
            company_pipeline = [
                {'$unwind': '$experience'},
                {'$group': {'_id': '$experience.company_name', 'count': {'$sum': 1}}},
                {'$sort': {'count': -1}},
                {'$limit': 10}
            ]
            top_companies = list(self.collection.aggregate(company_pipeline))
            
            education_pipeline = [
                {'$unwind': '$education'},
                {'$group': {'_id': '$education.college', 'count': {'$sum': 1}}},
                {'$sort': {'count': -1}},
                {'$limit': 10}
            ]
            top_schools = list(self.collection.aggregate(education_pipeline))
            
            return {
                'total_profiles': total_profiles,
                'today_profiles': today_profiles,
                'top_companies': top_companies,
                'top_schools': top_schools
            }
        except Exception as e:
            logger.error(f"Error getting statistics: {e}")
            return {'total_profiles': 0, 'today_profiles': 0, 'top_companies': [], 'top_schools': []}
    
    def delete_profile(self, profile_id):
        """Delete a profile by ID"""
        try:
            if self.collection is None:
                return False
                
            result = self.collection.delete_one({'_id': ObjectId(profile_id)})
            return result.deleted_count > 0
        except Exception as e:
            logger.error(f"Error deleting profile {profile_id}: {e}")
            return False

class ScrapingManager:
    """Handles scraping operations"""
    
    @staticmethod
    def validate_linkedin_url(url):
        """Validate if URL is a valid LinkedIn profile URL"""
        pattern = r'^https:\/\/www\.linkedin\.com\/in\/[a-zA-Z0-9\-]+\/?$'
        return re.match(pattern, url) is not None
    
    @staticmethod
    def run_scraper(url):
        """Run the actual scraper for a given URL"""
        try:
            result = run_linkedin_scraper(url)
            if isinstance(result, tuple) and len(result) >= 2:
                success, message = result[0], result[1]
            elif isinstance(result, bool):
                success, message = result, 'OK' if result else 'Failed'
            else:
                success, message = False, f'Unexpected scraper return: {repr(result)}'
            return success, message
        except Exception as e:
            logger.error(f"run_scraper exception: {e}")
            return False, str(e)

class ExportManager:
    """Handles data export operations with proper formatting"""
    
    def __init__(self, collection):
        self.collection = collection
    
    def clean_text(self, text):
        """Clean text data"""
        if text is None:
            return ""
        s = str(text)
        replacements = {
            'Â·': '·', '·': '-', '•': '•', '\xa0': ' ', '\u200b': '',
            '\r\n': ' ', '\r': ' ', '\n': ' '
        }
        for k, v in replacements.items():
            s = s.replace(k, v)
        s = re.sub(r'[\x00-\x1F\x7F]+', ' ', s)
        s = re.sub(r'\s+', ' ', s).strip()
        return s
    
    def extract_dates(self, duration_str):
        """Extract dates from duration string"""
        if not duration_str:
            return {"duration": "", "start_date": "", "end_date": "", "duration_months": ""}
        
        s = self.clean_text(duration_str)
        pattern = r'(\w{3}\s+\d{4})\s+to\s+((?:\w{3}\s+\d{4})|Present)\s*·?\s*(.+)?'
        match = re.search(pattern, s, re.IGNORECASE)
        
        if match:
            duration_text = self.clean_text(match.group(3) or "")
            duration_months = self.convert_duration_to_months(duration_text)
            
            return {
                "start_date": match.group(1),
                "end_date": match.group(2),
                "duration": duration_text,
                "duration_months": duration_months
            }
        
        # If no match, try to extract duration directly
        duration_months = self.convert_duration_to_months(s)
        return {"duration": s, "start_date": "", "end_date": "", "duration_months": duration_months}
    
    def convert_duration_to_months(self, duration_str):
        """Convert duration string to total months"""
        if not duration_str:
            return ""
        
        s = duration_str.lower()
        total_months = 0
        
        # Extract years
        year_match = re.search(r'(\d+)\s*yr', s)
        if year_match:
            total_months += int(year_match.group(1)) * 12
        
        # Extract months
        month_match = re.search(r'(\d+)\s*mo', s)
        if month_match:
            total_months += int(month_match.group(1))
        
        return str(total_months) if total_months > 0 else ""
    
    def extract_location(self, location_str):
        """Extract location details"""
        if not location_str:
            return {"city": "", "state": "", "country": "", "work_mode": ""}
        
        s = self.clean_text(location_str)
        work_mode = ""
        
        for mode in ["Remote", "On-site", "Hybrid", "Onsite"]:
            if mode.lower() in s.lower():
                work_mode = mode
                s = s.replace(mode, "").strip()
        
        s = re.sub(r'\s*·\s*', '', s).strip()
        parts = [p.strip() for p in s.split(',') if p.strip()]
        
        return {
            "city": parts[0] if len(parts) > 0 else "",
            "state": parts[1] if len(parts) > 1 else "",
            "country": parts[2] if len(parts) > 2 else (parts[-1] if parts else ""),
            "work_mode": work_mode
        }
    
    def export_to_excel(self, format_type='grouped', filters=None):
        """Export profiles to Excel with proper formatting"""
        try:
            if self.collection is None:
                return None, "Database not connected"
            
            query = filters if filters else {}
            profiles = list(self.collection.find(query))
            
            if not profiles:
                return None, "No profiles found to export"
            
            if format_type == 'grouped':
                return self._export_grouped_format(profiles)
            else:
                return self._export_flat_format(profiles)
                
        except Exception as e:
            logger.error(f"Error exporting to Excel: {e}")
            return None, str(e)
    
    def _export_grouped_format(self, profiles):
        """Export in grouped format with proper structure"""
        try:
            filename = f"linkedin_profiles_grouped_{datetime.now().strftime('%Y%m%d_%H%M%S')}.xlsx"
            filepath = os.path.join('exports', filename)
            
            basic_data = []
            exp_data = []
            edu_data = []
            proj_data = []
            cert_data = []
            awards_data = []
            courses_data = []
            
            for profile in profiles:
                name = self.clean_text(profile.get('name', ''))
                url = self.clean_text(profile.get('url', ''))
                
                # Basic Info - includes URL
                basic_data.append({
                    'name': name,
                    'url': url,
                    'headline': self.clean_text(profile.get('headline', '')),
                    'about': self.clean_text(profile.get('about', ''))[:500],
                    'last_updated': profile.get('last_updated', '')
                })
                
                # Experience - no URL, name only for first record
                experiences = profile.get('experience', [])
                for idx, exp in enumerate(experiences):
                    dates = self.extract_dates(exp.get('duration', ''))
                    location = self.extract_location(exp.get('location', ''))
                    
                    exp_data.append({
                        'name': name if idx == 0 else '',  # Name only for first record
                        'company_name': self.clean_text(exp.get('company_name', '')),
                        'designation': self.clean_text(exp.get('designation', '')),
                        'employment_type': self.clean_text(exp.get('role', '')),
                        'duration_months': dates['duration_months'],

                        'country': location['country'],
                        'work_mode': location['work_mode']
                    })

                educations = profile.get('education', [])
                for idx, edu in enumerate(educations):
                    dates = self.extract_dates(edu.get('duration', ''))

                        'name': name if idx == 0 else '',  # Name only for first record
                        'college': self.clean_text(edu.get('college', '')),
                        'degree': self.clean_text(edu.get('degree', '')),
                        'duration': dates['duration'],
                        'start_date': dates['start_date'],
                        'end_date': dates['end_date'],
                        'grade': self.clean_text(edu.get('grade', ''))
                    })

            # Write to Excel

    def _export_flat_format(self, profiles):
        """Export in flat format (single sheet)"""
        try:
            filename = f"linkedin_profiles_flat_{datetime.now().strftime('%Y%m%d_%H%M%S')}.xlsx"
            filepath = os.path.join('exports', filename)
            
            flat_data = []
            for profile in profiles:
                first_exp = profile.get('experience', [{}])[0] if profile.get('experience') else {}
                first_edu = profile.get('education', [{}])[0] if profile.get('education') else {}
                
                # Calculate total experience in months
                total_months = 0
                for exp in profile.get('experience', []):
                    dates = self.extract_dates(exp.get('duration', ''))
                    if dates['duration_months']:
                        total_months += int(dates['duration_months'])
                
                flat_data.append({
                    'Name': profile.get('name', ''),
                    'URL': profile.get('url', ''),
                    'Headline': profile.get('headline', ''),
                    'About': profile.get('about', '')[:300] if profile.get('about') else '',
                    'Current_Company': first_exp.get('company_name', ''),
                    'Current_Position': first_exp.get('designation', ''),
                    'Total_Experience_Months': total_months,
                    'Education_College': first_edu.get('college', ''),
                    'Education_Degree': first_edu.get('degree', ''),
                    'Total_Projects': len(profile.get('projects', [])),
                    'Total_Certifications': len(profile.get('licenses', [])),
                    'Last_Updated': profile.get('last_updated', '')
                })
            
            df = pd.DataFrame(flat_data)
            df.to_excel(filepath, index=False, engine='openpyxl')
            
            return filepath, None
            
        except Exception as e:
            return None, str(e)

@app.route('/api/scrape', methods=['POST'])
def api_scrape():
    """API endpoint to scrape a profile (creates new or updates existing)"""
    try:
        data = request.get_json()
        url = data.get('url', '').strip()
        
        if not url:
            return jsonify({'success': False, 'error': 'URL is required'}), 400
        
        if not ScrapingManager.validate_linkedin_url(url):
            return jsonify({'success': False, 'error': 'Invalid LinkedIn profile URL'}), 400
        
        # Check if profile exists
        is_update = False
        profile_id = None
        if collection is not None:
            existing = collection.find_one({'url': url})
            if existing:
                is_update = True
                profile_id = str(existing['_id'])
        
        # Run scraper (it will handle create or update)
        success, message = ScrapingManager.run_scraper(url)
        
        if success:
            if is_update:
                return jsonify({
                    'success': True, 
                    'message': 'Profile updated successfully',
                    'url': url,
                    'action': 'updated',
                    'profile_id': profile_id
                })
            else:
                # Get the newly created profile ID
                new_profile = collection.find_one({'url': url})

@app.route('/api/bulk-scrape', methods=['POST'])
def api_bulk_scrape():
    """API endpoint for bulk scraping"""
    try:
        data = request.get_json()
        urls = data.get('urls', [])
        
        if not urls:
            return jsonify({'success': False, 'error': 'No URLs provided'}), 400
        
        valid_urls = []
        invalid_urls = []
        existing_urls = []
        
        for url in urls:
            url = url.strip()
            if ScrapingManager.validate_linkedin_url(url):
                # Don't skip existing URLs anymore - they will be updated
                valid_urls.append(url)
                if collection is not None and collection.find_one({'url': url}):
                    existing_urls.append(url)
            else:
                invalid_urls.append(url)
        
        if not valid_urls:
            return jsonify({
                'success': False, 
                'error': 'No valid URLs to scrape',
                'invalid_urls': invalid_urls
            }), 400
        
        def bulk_scrape_background():
            for url in valid_urls:
                try:
                    success, message = ScrapingManager.run_scraper(url)
                    if success:
                        logger.info(f"Bulk scraped: {url}")
                    else:
                        logger.error(f"Failed bulk scraping {url}: {message}")
                except Exception as e:
                    logger.error(f"Bulk scraping error for {url}: {e}")
        
        thread = threading.Thread(target=bulk_scrape_background)
        thread.daemon = True
        thread.start()
        
        return jsonify({
            'success': True,
            'message': f'Bulk scraping started for {len(valid_urls)} profiles ({len(existing_urls)} will be updated)',
            'valid_urls': valid_urls,
            'existing_urls': existing_urls,
            'invalid_urls': invalid_urls
        })
        
    except Exception as e:
        logger.error(f"API bulk scrape error: {e}")
        return jsonify({'success': False, 'error': str(e)}), 500

def get_experience_tokens(driver, timeout=15):
    """Scroll and fetch all hidden text tokens from the Experience section."""
    driver.execute_script("window.scrollTo(0, document.body.scrollHeight * 0.35);")
    sleep(2)

    try:
        exp_section = WebDriverWait(driver, timeout).until(
            EC.presence_of_element_located((By.XPATH, "//section[.//span[text()='Experience']]"))
        )
    except TimeoutException:
        logger.info("ℹ️ No 'Experience' section found on this profile.")
        return []

    exp_html = exp_section.get_attribute("outerHTML")
    exp_soup = BeautifulSoup(exp_html, "lxml")
    tokens = [s.get_text(strip=True) for s in exp_soup.select("span.visually-hidden") if s.get_text(strip=True)]
    return tokens


def parse_experience_tokens(tokens):
    """Parse tokens into clean company_name, designation, duration, location"""
    if tokens and tokens[0].strip().lower() == "experience":
        tokens = tokens[1:]

    roles_set = {
        "Full-time", "Part-time", "Internship", "Contract", "Self-employed",
        "Freelance", "Apprenticeship", "Volunteer", "Temporary", "Seasonal"
    }

    def is_duration(s: str) -> bool:
        s2 = s.lower()
        return (" to " in s2 and ("present" in s2 or "mo" in s2 or "yr" in s2)) or (" yr" in s2) or (" mo" in s2)

    def is_location(s: str) -> bool:
        if is_duration(s): return False
        if s in roles_set: return False
        loc_keywords = ["india", "remote", "on-site", "hybrid", "onsite", "work from home"]
        if any(k in s.lower() for k in loc_keywords): return True
        return ("," in s and len(s.split()) >= 2)

    def clean_company(s: str) -> str:
        if not s: return ""
        for r in roles_set:
            s = s.replace("· " + r, "").replace(r, "")
        return s.strip()

    def looks_like_description(s: str, is_company=False) -> bool:
        if is_company: return False
        return (len(s.split()) > 15 or len(s) > 120 or s.lower().endswith((".png", ".jpg", ".jpeg")))

    result = []
    i, n = 0, len(tokens)

    while i < n:
        designation, company, dur, loc = "", "", "", ""

        if i < n:
            designation = tokens[i]; i += 1

        if i < n:
            if tokens[i] not in roles_set and not is_duration(tokens[i]) and not is_location(tokens[i]):
                company = tokens[i]; i += 1

        if i < n and is_duration(tokens[i]):
            dur = tokens[i]; i += 1

        if i < n and is_location(tokens[i]):
            loc = tokens[i]; i += 1

        # Fix: swap if designation looks like duration
        if is_duration(designation):
            dur, designation = designation, ""

        if looks_like_description(designation):
            designation = ""

        if company and any(kw in company.lower() for kw in ["pvt", "inc", "solutions", "system", "hr", "ltd"]):
            pass
        elif looks_like_description(company, is_company=True):
            company = ""

        if not designation and not company:
            continue

        result.append({
            "company_name": clean_company(company),
            "designation": designation,
            "duration": dur,
            "location": loc
        })

    clean_result = [r for r in result if not looks_like_description(r["duration"])]
    return clean_result


def run_linkedin_scraper(profile_url):
    """
    Scrapes a LinkedIn profile and saves/updates it in MongoDB.
    If profile exists, it will be updated with new data.
    """
    try:
        # MongoDB connection
        client = MongoClient(os.getenv("MONGO_URI", "mongodb://localhost:27017/"))
        db = client[os.getenv("DATABASE_NAME", "Linkedin")]
        collection = db[os.getenv("COLLECTION_NAME", "profile_data")]

        # Check if profile already exists
        existing_profile = collection.find_one({"url": profile_url})
        is_update = existing_profile is not None
        
        if is_update:
            logger.info(f"🔄 Profile exists, will update: {profile_url}")
        else:
            logger.info(f"🆕 New profile, will create: {profile_url}")

        # Chrome options
        chrome_options = Options()
        chrome_options.add_argument("--headless=new")
        chrome_options.add_argument("--no-sandbox")
        chrome_options.add_argument("--disable-dev-shm-usage")
        chrome_options.add_argument("--disable-blink-features=AutomationControlled")
        chrome_options.add_argument("--window-size=1920,1080")

        driver = webdriver.Chrome(options=chrome_options)
        wait = WebDriverWait(driver, 20)

        try:
            # --- LOGIN ---
            driver.get("https://www.linkedin.com/login")
            sleep(2)

            email = os.getenv("Email")
            password = os.getenv("Password")

            if not email or not password:
                raise ValueError("LinkedIn Email or Password not found in .env file")

            driver.find_element(By.ID, "username").send_keys(email)
            driver.find_element(By.ID, "password").send_keys(password, Keys.RETURN)
            sleep(5)

            # --- PROFILE LOAD ---
            driver.get(profile_url)
            wait.until(EC.presence_of_element_located((By.TAG_NAME, "body")))
            sleep(3)

            # Progressive scroll
            for _ in range(5):
                driver.execute_script("window.scrollBy(0, document.body.scrollHeight/5);")
                sleep(1)

            # Click "see more" buttons
            for btn_text in ["see more", "See more", "Show all"]:
                try:
                    buttons = driver.find_elements(By.XPATH, f"//button[contains(., '{btn_text}')]")
                    for btn in buttons[:3]:  # Limit to avoid infinite loops
                        try:
                            driver.execute_script("arguments[0].click();", btn)
                            sleep(1)
                        except:
                            pass
                except:
                    continue

            page_source = driver.page_source
            soup = BeautifulSoup(page_source, "lxml")

            # --- NAME ---
            try:
                name_elem = soup.find("h1", class_="text-heading-xlarge")
                if not name_elem:
                    name_elem = soup.find("h1")
                name = name_elem.get_text(strip=True) if name_elem else "N/A"
            except:
                name = "N/A"

            # --- HEADLINE ---
            try:
                headline = soup.find("div", class_="text-body-medium").get_text(strip=True)
            except:
                headline = "N/A"

            # --- ABOUT ---
            try:
                # Try clicking show more for about
                try:
                    driver.find_element(By.CLASS_NAME, "inline-show-more-text__button").click()
                    sleep(1)
                    page_source = driver.page_source
                    soup = BeautifulSoup(page_source, "lxml")
                except:
                    pass

                about_div = soup.find('div', {'class': 'display-flex ph5 pv3'})
                if not about_div:
                    about_div = soup.find('section', {'id': 'about'})
                about = about_div.get_text(strip=True) if about_div else "N/A"
            except:
                about = "N/A"

            # --- EXPERIENCE ---
            tokens = get_experience_tokens(driver)
            experience = parse_experience_tokens(tokens) if tokens else []

            # --- EDUCATION ---
            try:
                driver.execute_script("window.scrollTo(0, document.body.scrollHeight * 0.5);")
                sleep(2)

                edu_section = driver.find_elements(By.XPATH, "//section[.//span[text()='Education']]")
                if edu_section:
                    edu_html = edu_section[0].get_attribute("outerHTML")
                    edu_soup = BeautifulSoup(edu_html, "lxml")
                    edu_blocks = edu_soup.find_all("li", {"class": "artdeco-list__item"})
                    if not edu_blocks:
                        edu_blocks = edu_soup.find_all("li", {"class": "pvs-list__paged-list-item"})
                else:
                    edu_blocks = []

                education = []
                for block in edu_blocks:
                    spans = block.find_all("span", {"class": "visually-hidden"})
                    if not spans: continue

                    edu_dict = {}
                    if len(spans) >= 1:
                        edu_dict["college"] = spans[0].get_text(strip=True)
                    if len(spans) >= 2:
                        edu_dict["degree"] = spans[1].get_text(strip=True)
                    if len(spans) >= 3:
                        edu_dict["duration"] = spans[2].get_text(strip=True)
                    if len(spans) >= 4:
                        edu_dict["grade"] = spans[3].get_text(strip=True)

                    if edu_dict.get("college"):
                        education.append(edu_dict)
            except Exception as e:
                logger.warning(f"⚠️ Education error: {e}")
                education = []

            # --- PROJECTS ---
            try:
                driver.execute_script("window.scrollTo(0, document.body.scrollHeight * 0.6);")
                sleep(2)

                proj_section = driver.find_elements(By.XPATH, "//section[.//span[text()='Projects']]")
                if proj_section:
                    proj_html = proj_section[0].get_attribute("outerHTML")
                    proj_soup = BeautifulSoup(proj_html, "lxml")
                    items = proj_soup.find_all("li", {"class": "artdeco-list__item"})
                    if not items:
                        items = proj_soup.find_all("li", {"class": "pvs-list__paged-list-item"})
                else:
                    items = []

                projects = []
                for item in items:
                    spans = item.find_all('span', {'class': 'visually-hidden'})
                    filtered_spans = [
                        s.get_text(strip=True) for s in spans
                        if not any(keyword in s.get_text(strip=True).lower() 
                                 for keyword in ['connection', 'mutual', 'profile', 'view'])
                    ]

                    proj_dict = {}
                    if len(filtered_spans) > 0:
                        proj_dict['project_name'] = filtered_spans[0]
                    if len(filtered_spans) > 1:
                        proj_dict['duration'] = filtered_spans[1]
                    if len(filtered_spans) > 2:
                        proj_dict['description'] = filtered_spans[2]
                    if len(filtered_spans) > 3:
                        proj_dict['skills'] = filtered_spans[3]

                    if proj_dict.get('project_name'):
                        projects.append(proj_dict)
            except Exception as e:
                logger.warning(f"⚠️ Projects error: {e}")
                projects = []

            # --- LICENSES/CERTIFICATIONS ---
            try:
                driver.execute_script("window.scrollTo(0, document.body.scrollHeight * 0.7);")
                sleep(2)

                lic_section = driver.find_elements(By.XPATH, 
                    "//section[.//span[contains(text(), 'Licenses') or contains(text(), 'Certifications')]]")
                
                if lic_section:
                    lic_html = lic_section[0].get_attribute("outerHTML")
                    lic_soup = BeautifulSoup(lic_html, "lxml")
                    items = lic_soup.find_all("li", {"class": "artdeco-list__item"})
                    if not items:
                        items = lic_soup.find_all("li", {"class": "pvs-list__paged-list-item"})
                else:
                    items = []

                licenses = []
                for item in items:
                    spans = item.find_all('span', {'class': 'visually-hidden'})
                    filtered_spans = [
                        s.get_text(strip=True) for s in spans
                        if not any(keyword in s.get_text(strip=True).lower() 
                                 for keyword in ['connection', 'mutual', 'profile', 'view'])
                    ]

                    lic_dict = {}
                    if len(filtered_spans) > 0:
                        lic_dict['name'] = filtered_spans[0]
                    if len(filtered_spans) > 1:
                        lic_dict['institute'] = filtered_spans[1]
                    if len(filtered_spans) > 2:
                        lic_dict['issued_date'] = filtered_spans[2]

                    if lic_dict.get('name'):
                        licenses.append(lic_dict)
            except Exception as e:
                logger.warning(f"⚠️ Licenses error: {e}")
                licenses = []

            # --- COURSES ---
            try:
                courses_section = driver.find_elements(By.XPATH, 
                    "//section[.//span[contains(text(), 'Course')]]")
                
                courses = []
                if courses_section:
                    courses_html = courses_section[0].get_attribute("outerHTML")
                    courses_soup = BeautifulSoup(courses_html, "lxml")
                    items = courses_soup.find_all("li", {"class": "artdeco-list__item"})
                    if not items:
                        items = courses_soup.find_all("li", {"class": "pvs-list__paged-list-item"})

                    for item in items:
                        spans = item.find_all('span', {'class': 'visually-hidden'})
                        filtered_spans = [
                            s.get_text(strip=True) for s in spans
                            if not any(keyword in s.get_text(strip=True).lower() 
                                     for keyword in ['connection', 'mutual', 'profile'])
                        ]

                        if len(filtered_spans) >= 2:
                            courses.append({
                                'course_name': filtered_spans[0],
                                'associated_with': filtered_spans[1]
                            })
            except Exception as e:
                logger.warning(f"⚠️ Courses error: {e}")
                courses = []

            # --- HONORS & AWARDS ---
            try:
                awards_section = driver.find_elements(By.XPATH, 
                    "//section[.//span[contains(text(), 'Honors') or contains(text(), 'awards')]]")
                
                honors_awards = []
                if awards_section:
                    awards_html = awards_section[0].get_attribute("outerHTML")
                    awards_soup = BeautifulSoup(awards_html, "lxml")
                    items = awards_soup.find_all("li", {"class": "artdeco-list__item"})
                    if not items:
                        items = awards_soup.find_all("li", {"class": "pvs-list__paged-list-item"})

                    for item in items:
                        spans = [s.get_text(strip=True) for s in item.find_all("span", {"class": "visually-hidden"})]
                        filtered_spans = [
                            span for span in spans
                            if not any(x in span.lower() for x in ["connection", "studied at", "view profile"])
                        ]

                        if filtered_spans:
                            award_dict = {}
                            if len(filtered_spans) >= 1:
                                award_dict['award_name'] = filtered_spans[0]
                            if len(filtered_spans) >= 2:
                                parts = filtered_spans[1].split('·')
                                award_dict['issued_by'] = parts[0].replace('Issued by', '').strip()
                                if len(parts) > 1:
                                    award_dict['issued_date'] = parts[1].strip()
                            if len(filtered_spans) >= 3:
                                award_dict['associated_with'] = filtered_spans[2].replace('Associated with', '').strip()

                            honors_awards.append(award_dict)
            except Exception as e:
                logger.warning(f"⚠️ Awards error: {e}")
                honors_awards = []

            # --- PREPARE PROFILE DATA ---
            profile_data = {
                "name": name,
                "url": profile_url,
                "headline": headline,
                "about": about,
                "experience": experience,
                "education": education,
                "projects": projects,
                "licenses": licenses,
                "courses": courses,
                "honors_awards": honors_awards,
                "last_updated": datetime.now()
            }
            
            # Add scraped_at only for new profiles
            if not is_update:
                profile_data["scraped_at"] = datetime.now()

            # --- SAVE OR UPDATE TO MONGODB ---
            if is_update:
                # Update existing profile
                collection.update_one(
                    {"url": profile_url},
                    {"$set": profile_data}
                )
                logger.info(f"🔄 Profile updated: {name}")
                return True, "Profile updated successfully"
            else:
                # Insert new profile
                collection.insert_one(profile_data)
                logger.info(f"✅ Profile saved: {name}")
                return True, "Profile scraped successfully"

        finally:
            driver.quit()

    except Exception as e:
        logger.error(f"❌ Error scraping {profile_url}: {e}")
        return False, str(e)

    finally:
        if "client" in locals():
            client.close()

        // Export profiles function
        async function exportProfiles(format = 'grouped') {
            try {
                const response = await fetch('/api/export', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify({ format: format })
                });

                if (response.ok) {
                    const blob = await response.blob();
                    const url = window.URL.createObjectURL(blob);
                    const a = document.createElement('a');
                    a.href = url;
                    a.download = `linkedin_profiles_${format}_${new Date().toISOString().split('T')[0]}.xlsx`;
                    document.body.appendChild(a);
                    a.click();
                    window.URL.revokeObjectURL(url);
                    document.body.removeChild(a);
                    
                    showAlert('success', 'Export completed successfully!');
                } else {
                    const error = await response.json();
                    showAlert('danger', 'Export failed: ' + (error.error || 'Unknown error'));
                }
            } catch (error) {
                showAlert('danger', 'Export failed: ' + error.message);
            }
        }

        // Refresh statistics
        async function refreshStats() {
            try {
                const response = await fetch('/api/stats');
                if (response.ok) {
                    location.reload(); // Simple refresh for now
                } else {
                    showAlert('warning', 'Failed to refresh statistics');
                }
            } catch (error) {
                showAlert('danger', 'Error refreshing statistics: ' + error.message);
            }
        }

        // Show alert messages
        function showAlert(type, message) {
            const alertDiv = document.createElement('div');
            alertDiv.className = `alert alert-${type} alert-dismissible fade show`;
            alertDiv.innerHTML = `
                ${message}
                <button type="button" class="btn-close" data-bs-dismiss="alert"></button>
            `;
            
            const container = document.querySelector('.container');
            container.insertBefore(alertDiv, container.firstChild);
            
            // Auto-dismiss after 5 seconds
            setTimeout(() => {
                if (alertDiv.parentNode) {
                    alertDiv.remove();
                }
            }, 5000);
        }

        // Auto-refresh stats every 5 minutes
        setInterval(refreshStats, 5 * 60 * 1000);
    </script>

        // File upload handling
        function handleFileUpload(event) {
            const file = event.target.files[0];
            if (!file) return;
            
            const reader = new FileReader();
            reader.onload = function(e) {
                const text = e.target.result;
                const urls = text.split('\n')
                    .map(url => url.trim())
                    .filter(url => url && validateLinkedInUrl(url));
                
                if (urls.length > 0) {
                    uploadedUrls = urls;
                    displayUrlList(urls);
                    showAlert('success', `Loaded ${urls.length} valid LinkedIn URLs`);
                } else {
                    showAlert('danger', 'No valid LinkedIn URLs found in file');
                }
            };
            reader.readAsText(file);
        }

        // Drag and drop handling
        const uploadArea = document.getElementById('uploadArea');
        
        uploadArea.addEventListener('dragover', function(e) {
            e.preventDefault();
            uploadArea.classList.add('dragover');
        });
        
        uploadArea.addEventListener('dragleave', function(e) {
            e.preventDefault();
            uploadArea.classList.remove('dragover');
        });
        
        uploadArea.addEventListener('drop', function(e) {
            e.preventDefault();
            uploadArea.classList.remove('dragover');
            
            const files = e.dataTransfer.files;
            if (files.length > 0) {
                const file = files[0];
                if (file.type === 'text/plain' || file.name.endsWith('.csv') || file.name.endsWith('.txt')) {
                    const fakeEvent = { target: { files: [file] } };
                    handleFileUpload(fakeEvent);
                } else {
                    showAlert('danger', 'Please drop a .txt or .csv file');
                }
            }
        });

        // Display URL list
        function displayUrlList(urls) {
            const container = document.getElementById('urlListContainer');
            const urlList = document.getElementById('urlList');
            const urlCount = document.getElementById('urlCount');

                    </button>
                </div>
            `).join('');
            
            container.classList.remove('d-none');
        }

        // Remove URL from list
        function removeUrl(index) {
            uploadedUrls.splice(index, 1);
            if (uploadedUrls.length > 0) {
                displayUrlList(uploadedUrls);
            } else {
                document.getElementById('urlListContainer').classList.add('d-none');
            }
        }

        // Bulk scraping
        async function startBulkScraping() {
            let urls = uploadedUrls.slice();
            
            // Add manual URLs
            const manualUrls = document.getElementById('bulkUrls').value
                .split('\n')
                .map(url => url.trim())
                .filter(url => url && validateLinkedInUrl(url));
            
            urls = [...urls, ...manualUrls];
            
            if (urls.length === 0) {
                showAlert('danger', 'Please provide at least one valid LinkedIn URL');
                return;
            }
            
            const button = document.getElementById('bulkScrapeBtn');
            const spinner = document.getElementById('bulkSpinner');
            const icon = document.getElementById('bulkIcon');
            const buttonText = document.getElementById('bulkButtonText');
            const progressContainer = document.getElementById('bulkProgress');
            const resultsDiv = document.getElementById('bulkResults');
            
            // Show loading state
            button.disabled = true;
            spinner.classList.remove('d-none');
            icon.classList.add('d-none');
            buttonText.textContent = 'Starting Bulk Scraping...';
            progressContainer.style.display = 'block';
            resultsDiv.innerHTML = '';
            
            try {
                const response = await fetch('/api/bulk-scrape', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify({ urls: urls })
                });
                
                const result = await response.json();
                
                if (result.success) {
                    resultsDiv.innerHTML = `
                        <div class="alert alert-success">
                            <i class="fas fa-check-circle me-2"></i>
                            Bulk scraping started for ${result.valid_urls.length} profiles!
                            ${result.invalid_urls.length > 0 ? 
                                `<br><small class="text-muted">${result.invalid_urls.length} invalid URLs skipped</small>` 
                                : ''
                            }
                        </div>
                    `;
                    
                    // Clear forms
                    document.getElementById('bulkUrls').value = '';
                    uploadedUrls = [];
                    document.getElementById('urlListContainer').classList.add('d-none');
                    
                    // Simulate progress for bulk operation
                    simulateBulkProgress(result.valid_urls.length);
                    
                } else {
                    resultsDiv.innerHTML = `
                        <div class="alert alert-danger">
                            <i class="fas fa-exclamation-circle me-2"></i>
                            ${result.error}
                        </div>
                    `;
                }
                
            } catch (error) {
                resultsDiv.innerHTML = `
                    <div class="alert alert-danger">
                        <i class="fas fa-exclamation-circle me-2"></i>
                        Network error: ${error.message}
                    </div>
                `;
            } finally {
                // Reset button state
                button.disabled = false;
                spinner.classList.add('d-none');
                icon.classList.remove('d-none');
                buttonText.textContent = 'Start Bulk Scraping';
            }
        }

        // Simulate bulk progress
        function simulateBulkProgress(totalUrls) {
            let completed = 0;
            const progressInterval = setInterval(() => {
                completed += Math.random() * 2;
                if (completed >= totalUrls) {
                    completed = totalUrls;
                    clearInterval(progressInterval);
                    
                    setTimeout(() => {
                        document.getElementById('bulkProgress').style.display = 'none';
                    }, 3000);
                }
                
                const progress = (completed / totalUrls) * 100;
                updateProgress('bulk', progress, `${Math.floor(completed)}/${totalUrls}`);
            }, 2000);
        }

        // Update progress bars
        function updateProgress(type, percentage, text = null) {
            const progressBar = document.getElementById(`${type}ProgressBar`);
            const progressText = document.getElementById(`${type}ProgressText`);
            
            progressBar.style.width = `${percentage}%`;
            
            if (text) {
                progressText.textContent = text;
            } else {
                progressText.textContent = `${Math.round(percentage)}%`;
            }
        }

        // Validate LinkedIn URL
        function validateLinkedInUrl(url) {
            const pattern = /^https:\/\/www\.linkedin\.com\/in\/[a-zA-Z0-9\-]+\/?$/;
            return pattern.test(url);
        }

        // Show alert messages
        function showAlert(type, message) {
            const alertDiv = document.createElement('div');
            alertDiv.className = `alert alert-${type} alert-dismissible fade show`;
            alertDiv.innerHTML = `
                ${message}
                <button type="button" class="btn-close" data-bs-dismiss="alert"></button>
            `;
            
            const container = document.querySelector('.container');
            container.insertBefore(alertDiv, container.children[1]);
            
            // Auto-dismiss after 5 seconds
            setTimeout(() => {
                if (alertDiv.parentNode) {
                    alertDiv.remove();
                }
            }, 5000);
        }

class ProfileManager:
    """Handles all profile-related database operations"""
    
    def __init__(self, collection):
        self.collection = collection
    
    def get_all_profiles(self, limit=50, skip=0):
        """Get all profiles with pagination"""
        try:
            if self.collection is None:
                return []
                
            profiles = list(self.collection.find({}, {
                'name': 1, 'headline': 1, 'url': 1, 'last_updated': 1,
                'experience': {'$slice': 1}, 'education': {'$slice': 1}
            }).sort('last_updated', -1).skip(skip).limit(limit))
            
            for profile in profiles:
                profile['_id'] = str(profile['_id'])
                if 'last_updated' in profile and profile['last_updated']:
                    if isinstance(profile['last_updated'], datetime):
                        profile['last_updated'] = profile['last_updated'].strftime('%Y-%m-%d %H:%M:%S')
            
            return profiles
        except Exception as e:
            logger.error(f"Error fetching profiles: {e}")
            return []
    
    def get_profile_by_id(self, profile_id):
        """Get detailed profile by ID"""
        try:
            if self.collection is None:
                return None
                
            profile = self.collection.find_one({'_id': ObjectId(profile_id)})
            if profile:
                profile['_id'] = str(profile['_id'])
                if 'last_updated' in profile and profile['last_updated']:
                    if isinstance(profile['last_updated'], datetime):
                        profile['last_updated'] = profile['last_updated'].strftime('%Y-%m-%d %H:%M:%S')
            return profile
        except Exception as e:
            logger.error(f"Error fetching profile {profile_id}: {e}")
            return None
    
    def search_profiles(self, query, limit=20):
        """Search profiles by name, headline, or company"""
        try:
            if self.collection is None:
                return []
                
            search_filter = {
                '$or': [
                    {'name': {'$regex': query, '$options': 'i'}},
                    {'headline': {'$regex': query, '$options': 'i'}},
                    {'experience.company_name': {'$regex': query, '$options': 'i'}}
                ]
            }
            
            profiles = list(self.collection.find(search_filter, {
                'name': 1, 'headline': 1, 'url': 1, 'last_updated': 1
            }).limit(limit))
            
            for profile in profiles:
                profile['_id'] = str(profile['_id'])
            
            return profiles
        except Exception as e:
            logger.error(f"Error searching profiles: {e}")
            return []
    
    def get_statistics(self):
        """Get database statistics"""
        try:
            if self.collection is None:
                return {'total_profiles': 0, 'today_profiles': 0, 'top_companies': [], 'top_schools': []}
                
            total_profiles = self.collection.count_documents({})
            
            today_start = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
            today_profiles = self.collection.count_documents({
                'last_updated': {'$gte': today_start}
            })
            
            company_pipeline = [
                {'$unwind': '$experience'},
                {'$group': {'_id': '$experience.company_name', 'count': {'$sum': 1}}},
                {'$sort': {'count': -1}},
                {'$limit': 10}
            ]
            top_companies = list(self.collection.aggregate(company_pipeline))
            
            education_pipeline = [
                {'$unwind': '$education'},
                {'$group': {'_id': '$education.college', 'count': {'$sum': 1}}},
                {'$sort': {'count': -1}},
                {'$limit': 10}
            ]
            top_schools = list(self.collection.aggregate(education_pipeline))
            
            return {
                'total_profiles': total_profiles,
                'today_profiles': today_profiles,
                'top_companies': top_companies,
                'top_schools': top_schools
            }
        except Exception as e:
            logger.error(f"Error getting statistics: {e}")
            return {'total_profiles': 0, 'today_profiles': 0, 'top_companies': [], 'top_schools': []}
    
    def delete_profile(self, profile_id):
        """Delete a profile by ID"""
        try:
            if self.collection is None:
                return False
                
            result = self.collection.delete_one({'_id': ObjectId(profile_id)})
            return result.deleted_count > 0
        except Exception as e:
            logger.error(f"Error deleting profile {profile_id}: {e}")
            return False

class ScrapingManager:
    """Handles scraping operations"""
    
    @staticmethod
    def validate_linkedin_url(url):
        """Validate if URL is a valid LinkedIn profile URL"""
        pattern = r'^https:\/\/www\.linkedin\.com\/in\/[a-zA-Z0-9\-]+\/?$'
        return re.match(pattern, url) is not None
    
    @staticmethod
    def run_scraper(url):
        """Run the actual scraper for a given URL"""
        try:
            result = run_linkedin_scraper(url)
            if isinstance(result, tuple) and len(result) >= 2:
                success, message = result[0], result[1]
            elif isinstance(result, bool):
                success, message = result, 'OK' if result else 'Failed'
            else:
                success, message = False, f'Unexpected scraper return: {repr(result)}'
            return success, message
        except Exception as e:
            logger.error(f"run_scraper exception: {e}")
            return False, str(e)

class ExportManager:

    def __init__(self, collection):

    def clean_text(self, text):
        """Clean text data"""
        if text is None:
            return ""
        s = str(text)
        replacements = {
            'Â·': '·', '·': '-', '•': '•', '\xa0': ' ', '\u200b': '',
            '\r\n': ' ', '\r': ' ', '\n': ' '
        }
        for k, v in replacements.items():
            s = s.replace(k, v)
        s = re.sub(r'[\x00-\x1F\x7F]+', ' ', s)
        s = re.sub(r'\s+', ' ', s).strip()
        return s
    
    def extract_dates(self, duration_str):
        """Extract dates from duration string"""
        if not duration_str:
            return {"duration": "", "start_date": "", "end_date": "", "duration_months": ""}
        
        s = self.clean_text(duration_str)
        pattern = r'(\w{3}\s+\d{4})\s+to\s+((?:\w{3}\s+\d{4})|Present)\s*·?\s*(.+)?'
        match = re.search(pattern, s, re.IGNORECASE)
        
        if match:
            duration_text = self.clean_text(match.group(3) or "")
            duration_months = self.convert_duration_to_months(duration_text)
            
            return {
                "start_date": match.group(1),
                "end_date": match.group(2),
                "duration": duration_text,
                "duration_months": duration_months
            }
        
        # If no match, try to extract duration directly
        duration_months = self.convert_duration_to_months(s)
        return {"duration": s, "start_date": "", "end_date": "", "duration_months": duration_months}
    
    def convert_duration_to_months(self, duration_str):
        """Convert duration string to total months"""
        if not duration_str:
            return ""
        
        s = duration_str.lower()
        total_months = 0
        
        # Extract years
        year_match = re.search(r'(\d+)\s*yr', s)
        if year_match:
            total_months += int(year_match.group(1)) * 12
        
        # Extract months
        month_match = re.search(r'(\d+)\s*mo', s)
        if month_match:
            total_months += int(month_match.group(1))
        
        return str(total_months) if total_months > 0 else ""
    
    def extract_location(self, location_str):
        """Extract location details"""
        if not location_str:
            return {"city": "", "state": "", "country": "", "work_mode": ""}
        
        s = self.clean_text(location_str)
        work_mode = ""
        
        for mode in ["Remote", "On-site", "Hybrid", "Onsite"]:
            if mode.lower() in s.lower():
                work_mode = mode
                s = s.replace(mode, "").strip()
        
        s = re.sub(r'\s*·\s*', '', s).strip()
        parts = [p.strip() for p in s.split(',') if p.strip()]
        
        return {
            "city": parts[0] if len(parts) > 0 else "",
            "state": parts[1] if len(parts) > 1 else "",
            "country": parts[2] if len(parts) > 2 else (parts[-1] if parts else ""),
            "work_mode": work_mode
        }
    
    def export_to_excel(self, format_type='grouped', filters=None):

            if self.collection is None:
                return None, "Database not connected"
            
            query = filters if filters else {}

                return None, "No profiles found to export"

                return self._export_grouped_format(profiles)
            else:
                return self._export_flat_format(profiles)
                
        except Exception as e:
            logger.error(f"Error exporting to Excel: {e}")
            return None, str(e)

    def _export_flat_format(self, profiles):
        """Export in flat format (single sheet)"""
        try:
            filename = f"linkedin_profiles_flat_{datetime.now().strftime('%Y%m%d_%H%M%S')}.xlsx"
            filepath = os.path.join('exports', filename)
            
            flat_data = []
            for profile in profiles:
                first_exp = profile.get('experience', [{}])[0] if profile.get('experience') else {}
                first_edu = profile.get('education', [{}])[0] if profile.get('education') else {}
                
                # Calculate total experience in months
                total_months = 0
                for exp in profile.get('experience', []):
                    dates = self.extract_dates(exp.get('duration', ''))
                    if dates['duration_months']:
                        total_months += int(dates['duration_months'])
                
                flat_data.append({
                    'Name': profile.get('name', ''),
                    'URL': profile.get('url', ''),
                    'Headline': profile.get('headline', ''),
                    'About': profile.get('about', '')[:300] if profile.get('about') else '',
                    'Current_Company': first_exp.get('company_name', ''),
                    'Current_Position': first_exp.get('designation', ''),
                    'Total_Experience_Months': total_months,
                    'Education_College': first_edu.get('college', ''),
                    'Education_Degree': first_edu.get('degree', ''),
                    'Total_Projects': len(profile.get('projects', [])),
                    'Total_Certifications': len(profile.get('licenses', [])),
                    'Last_Updated': profile.get('last_updated', '')
                })
            
            df = pd.DataFrame(flat_data)
            df.to_excel(filepath, index=False, engine='openpyxl')
            
            return filepath, None
            
        except Exception as e:
            return None, str(e)

# Helper function for analytics
def calculate_profile_completeness(profile):
    """Calculate profile completeness percentage"""
    score = 0
    total_points = 10
    
    # Name (mandatory)
    if profile.get('name') and profile.get('name') != 'N/A':
        score += 1
    
    # Headline
    if profile.get('headline') and profile.get('headline') != 'N/A':
        score += 1
    
    # About section
    if profile.get('about') and profile.get('about') != 'N/A' and len(profile.get('about', '')) > 50:
        score += 1
    
    # Experience (at least one)
    if profile.get('experience') and len(profile.get('experience', [])) > 0:
        score += 2
    
    # Education (at least one)
    if profile.get('education') and len(profile.get('education', [])) > 0:
        score += 2
    
    # Projects
    if profile.get('projects') and len(profile.get('projects', [])) > 0:
        score += 1
    
    # Certifications
    if profile.get('licenses') and len(profile.get('licenses', [])) > 0:
        score += 1
    
    # Courses or Awards
    if (profile.get('courses') and len(profile.get('courses', [])) > 0) or \
       (profile.get('honors_awards') and len(profile.get('honors_awards', [])) > 0):
        score += 1
    
    return round((score / total_points) * 100, 1)

    except Exception as e:
        logger.error(f"Profiles page error: {e}")
        return render_template('profiles.html', profiles=[], current_page=1, total_pages=1, search_query='')

@app.route('/api/scrape', methods=['POST'])
def api_scrape():
    """API endpoint to scrape a profile (creates new or updates existing)"""
    try:
        data = request.get_json()
        url = data.get('url', '').strip()
        
        if not url:
            return jsonify({'success': False, 'error': 'URL is required'}), 400
        
        if not ScrapingManager.validate_linkedin_url(url):
            return jsonify({'success': False, 'error': 'Invalid LinkedIn profile URL'}), 400
        
        # Check if profile exists
        is_update = False
        profile_id = None
        if collection is not None:
            existing = collection.find_one({'url': url})
            if existing:
                is_update = True
                profile_id = str(existing['_id'])
        
        # Run scraper (it will handle create or update)
        success, message = ScrapingManager.run_scraper(url)
        
        if success:
            if is_update:
                return jsonify({
                    'success': True, 
                    'message': 'Profile updated successfully',
                    'url': url,
                    'action': 'updated',
                    'profile_id': profile_id
                })
            else:
                # Get the newly created profile ID
                new_profile = collection.find_one({'url': url})
                new_profile_id = str(new_profile['_id']) if new_profile else None
                
                return jsonify({
                    'success': True, 
                    'message': 'Profile scraped and saved successfully',
                    'url': url,
                    'action': 'created',
                    'profile_id': new_profile_id
                })
        else:
            return jsonify({
                'success': False,
                'error': message
            }), 500
        
    except Exception as e:
        logger.error(f"API scrape error: {e}")
        return jsonify({'success': False, 'error': str(e)}), 500

@app.route('/api/bulk-scrape', methods=['POST'])
def api_bulk_scrape():
    """API endpoint for bulk scraping"""
    try:
        data = request.get_json()
        urls = data.get('urls', [])
        
        if not urls:
            return jsonify({'success': False, 'error': 'No URLs provided'}), 400
        
        valid_urls = []
        invalid_urls = []
        existing_urls = []
        
        for url in urls:
            url = url.strip()
            if ScrapingManager.validate_linkedin_url(url):
                # Don't skip existing URLs anymore - they will be updated
                valid_urls.append(url)
                if collection is not None and collection.find_one({'url': url}):
                    existing_urls.append(url)
            else:
                invalid_urls.append(url)
        
        if not valid_urls:
            return jsonify({
                'success': False, 
                'error': 'No valid URLs to scrape',
                'invalid_urls': invalid_urls
            }), 400
        
        def bulk_scrape_background():
            for url in valid_urls:
                try:
                    success, message = ScrapingManager.run_scraper(url)
                    if success:
                        logger.info(f"Bulk scraped: {url}")
                    else:
                        logger.error(f"Failed bulk scraping {url}: {message}")
                except Exception as e:
                    logger.error(f"Bulk scraping error for {url}: {e}")
        
        thread = threading.Thread(target=bulk_scrape_background)
        thread.daemon = True
        thread.start()
        
        return jsonify({
            'success': True,
            'message': f'Bulk scraping started for {len(valid_urls)} profiles ({len(existing_urls)} will be updated)',
            'valid_urls': valid_urls,
            'existing_urls': existing_urls,
            'invalid_urls': invalid_urls
        })
        
    except Exception as e:
        logger.error(f"API bulk scrape error: {e}")
        return jsonify({'success': False, 'error': str(e)}), 500

@app.route('/api/advanced-analytics', methods=['GET'])
def api_advanced_analytics():
    """API endpoint for advanced analytics data"""
    try:
        if collection is None:
            return jsonify({'success': False, 'error': 'Database not connected'}), 500
        
        # Get all profiles for analysis
        all_profiles = list(collection.find({}))
        
        # Calculate experience distribution (in months)
        experience_distribution = []
        exp_ranges = [
            {'range': '0-12 months', 'min': 0, 'max': 12, 'count': 0},
            {'range': '1-3 years', 'min': 13, 'max': 36, 'count': 0},
            {'range': '3-5 years', 'min': 37, 'max': 60, 'count': 0},
            {'range': '5-10 years', 'min': 61, 'max': 120, 'count': 0},
            {'range': '10+ years', 'min': 121, 'max': 999999, 'count': 0}
        ]
        
        for profile in all_profiles:
            total_months = 0
            for exp in profile.get('experience', []):
                duration = exp.get('duration', '')
                # Extract months from duration
                years_match = re.search(r'(\d+)\s*yr', duration.lower())
                months_match = re.search(r'(\d+)\s*mo', duration.lower())
                
                if years_match:
                    total_months += int(years_match.group(1)) * 12
                if months_match:
                    total_months += int(months_match.group(1))
            
            # Categorize into ranges
            for exp_range in exp_ranges:
                if exp_range['min'] <= total_months <= exp_range['max']:
                    exp_range['count'] += 1
                    break
        
        experience_distribution = [{'range': r['range'], 'count': r['count']} for r in exp_ranges]
        
        # Top certifications
        cert_counter = {}
        for profile in all_profiles:
            for cert in profile.get('licenses', []):
                cert_name = cert.get('name', '')
                if cert_name:
                    cert_counter[cert_name] = cert_counter.get(cert_name, 0) + 1
        
        top_certifications = [
            {'name': name, 'count': count} 
            for name, count in sorted(cert_counter.items(), key=lambda x: x[1], reverse=True)[:10]
        ]
        
        # Profiles with/without projects
        profiles_with_projects = sum(1 for p in all_profiles if p.get('projects'))
        
        # Top locations
        location_counter = {}
        for profile in all_profiles:
            for exp in profile.get('experience', []):
                location = exp.get('location', '')
                if location:
                    # Extract country or city
                    parts = location.split(',')
                    if parts:
                        loc = parts[-1].strip() if len(parts) > 1 else parts[0].strip()
                        location_counter[loc] = location_counter.get(loc, 0) + 1
        
        top_locations = [
            {'location': loc, 'count': count}
            for loc, count in sorted(location_counter.items(), key=lambda x: x[1], reverse=True)[:8]
        ]
        
        # Hourly activity (when profiles were scraped)
        hourly_activity = [{'hour': i, 'count': 0} for i in range(24)]
        for profile in all_profiles:
            scraped_at = profile.get('scraped_at') or profile.get('last_updated')
            if scraped_at and isinstance(scraped_at, datetime):
                hour = scraped_at.hour
                hourly_activity[hour]['count'] += 1
        
        # Experience levels
        experience_levels = []
        total = len(all_profiles)
        for exp_range in exp_ranges:
            percentage = round((exp_range['count'] / total * 100), 1) if total > 0 else 0
            experience_levels.append({
                'level': exp_range['range'],
                'count': exp_range['count'],
                'percentage': percentage
            })
        
        # Unique companies and schools
        unique_companies = len(set(
            exp.get('company_name', '') 
            for p in all_profiles 
            for exp in p.get('experience', [])
            if exp.get('company_name')
        ))
        
        unique_schools = len(set(
            edu.get('college', '')
            for p in all_profiles
            for edu in p.get('education', [])
            if edu.get('college')
        ))
        
        analytics_data = {
            'success': True,
            'analytics': {
                'total_profiles': len(all_profiles),
                'experience_distribution': experience_distribution,
                'top_certifications': top_certifications,
                'profiles_with_projects': profiles_with_projects,
                'top_locations': top_locations,
                'hourly_activity': [h for h in hourly_activity if h['count'] > 0],
                'experience_levels': experience_levels,
                'unique_companies': unique_companies,
                'unique_schools': unique_schools
            }
        }
        
        return jsonify(analytics_data)
        
    except Exception as e:
        logger.error(f"Advanced analytics error: {e}")
        return jsonify({'success': False, 'error': str(e)}), 500

@app.template_filter('datetime_format')
def datetime_format(value, format='%Y-%m-%d %H:%M'):
    if isinstance(value, str):
        try:
            value = datetime.strptime(value, '%Y-%m-%d %H:%M:%S')
        except:
            return value
    return value.strftime(format) if value else ''

@app.template_filter('truncate_text')
def truncate_text(text, length=100):
    if not text:
        return ''
    return text[:length] + '...' if len(text) > length else text

@app.template_filter('format_number')
def format_number(value):
    try:
        return f"{int(value):,}"
    except:
        return value

@app.context_processor
def inject_stats():
    try:
        if collection is not None:
            basic_stats = {
                'total_profiles': collection.count_documents({}),
                'today_profiles': collection.count_documents({
                    'last_updated': {'$gte': datetime.now().replace(hour=0, minute=0, second=0)}
                })
            }
        else:
            basic_stats = {'total_profiles': 0, 'today_profiles': 0}
        return dict(global_stats=basic_stats)
    except:
        return dict(global_stats={'total_profiles': 0, 'today_profiles': 0})
